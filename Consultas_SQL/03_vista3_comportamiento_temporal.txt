================================================================================
VISTA 3 — COMPORTAMIENTO TEMPORAL (2021–2026)
================================================================================
Pregunta: ¿Cómo ha evolucionado el volumen de trabajadores, el ingreso neto y
la intensidad horaria semanal a lo largo de los seis años analizados?

Dimensiones: dim_tiempo (anio, trimestre, mes)
             dim_plataforma (tipo_plataforma)
Medidas:     COUNT(DISTINCT id_trabajador), AVG(ingreso_neto_mensual_cop),
             AVG(horas_semanales).

--------------------------------------------------------------------------------
1. INDICADORES GLOBALES (KPIs)
--------------------------------------------------------------------------------
SELECT COUNT(DISTINCT f.id_trabajador)                     AS volumen_total,
       ROUND(AVG(f.horas_semanales)::numeric, 1)           AS jornada_promedio_h
FROM fact_gig_economy f;
-- Esperado: 12.350 trabajadores · 46,4 h/semana.


--------------------------------------------------------------------------------
2. VOLUMEN DE TRABAJADORES POR AÑO
--------------------------------------------------------------------------------
SELECT ti.anio,
       COUNT(DISTINCT f.id_trabajador) AS trabajadores
FROM fact_gig_economy f
JOIN dim_tiempo ti ON ti.id_tiempo = f.id_tiempo
GROUP BY ti.anio
ORDER BY ti.anio;
-- Esperado: 2021 2.198 · 2022 2.292 · 2023 2.280 · 2024 2.223 ·
--           2025 2.248 · 2026 1.109 (corte parcial enero–junio).


--------------------------------------------------------------------------------
3. DRILL-DOWN: VOLUMEN DE TRABAJADORES POR TRIMESTRE
--------------------------------------------------------------------------------
SELECT ti.anio,
       ti.trimestre,
       COUNT(DISTINCT f.id_trabajador) AS trabajadores
FROM fact_gig_economy f
JOIN dim_tiempo ti ON ti.id_tiempo = f.id_tiempo
GROUP BY ti.anio, ti.trimestre
ORDER BY ti.anio, ti.trimestre;
-- Esperado: ~530–600 trabajadores por trimestre (sin estacionalidad relevante).


--------------------------------------------------------------------------------
4. INGRESO NETO PROMEDIO POR AÑO
--------------------------------------------------------------------------------
SELECT ti.anio,
       ROUND(AVG(f.ingreso_neto_mensual_cop)::numeric, 0) AS ingreso_promedio_cop,
       ROUND(AVG(f.ingreso_neto_hora_usd)::numeric, 2)    AS tarifa_promedio_usd_h,
       ROUND(AVG(f.horas_semanales)::numeric, 1)           AS jornada_promedio_h
FROM fact_gig_economy f
JOIN dim_tiempo ti ON ti.id_tiempo = f.id_tiempo
GROUP BY ti.anio
ORDER BY ti.anio;
-- Esperado: ingreso nominal prácticamente plano (4.169.353 en 2023 a 4.451.975 en 2025).


--------------------------------------------------------------------------------
5. VARIACIÓN DEL INGRESO NETO 2021 vs 2025 (periodos equivalentes)
--------------------------------------------------------------------------------
WITH base AS (
    SELECT ti.anio, AVG(f.ingreso_neto_mensual_cop) AS ingreso
    FROM fact_gig_economy f
    JOIN dim_tiempo ti ON ti.id_tiempo = f.id_tiempo
    WHERE ti.anio IN (2021, 2025)
    GROUP BY ti.anio
)
SELECT MAX(ingreso) FILTER (WHERE anio = 2021)                     AS ingreso_2021,
       MAX(ingreso) FILTER (WHERE anio = 2025)                     AS ingreso_2025,
       ROUND(100.0 * (MAX(ingreso) FILTER (WHERE anio = 2025) -
                      MAX(ingreso) FILTER (WHERE anio = 2021)) /
                      MAX(ingreso) FILTER (WHERE anio = 2021), 1)  AS variacion_pct
FROM base;
-- Esperado: +3,0% (4.320.350 -> 4.451.975 COP).


--------------------------------------------------------------------------------
6. JORNADA SEMANAL PROMEDIO POR AÑO x MODELO DE PLATAFORMA
--------------------------------------------------------------------------------
SELECT ti.anio,
       p.tipo_plataforma,
       ROUND(AVG(f.horas_semanales)::numeric, 1) AS jornada_promedio_h
FROM fact_gig_economy f
JOIN dim_tiempo ti ON ti.id_tiempo = f.id_tiempo
JOIN dim_plataforma p ON p.id_plataforma = f.id_plataforma
GROUP BY ti.anio, p.tipo_plataforma
ORDER BY ti.anio, p.tipo_plataforma;
-- Esperado: Física 52,1–53,4 h (siempre sobre el límite legal de 47 h) ·
--           En línea 33,6–34,6 h.


--------------------------------------------------------------------------------
7. MATRIZ: AÑO x MODELO DE PLATAFORMA (medida = ingreso neto promedio COP)
--------------------------------------------------------------------------------
SELECT ti.anio,
       ROUND(AVG(f.ingreso_neto_mensual_cop) FILTER (
             WHERE p.tipo_plataforma = 'Fisica / Basada en Ubicacion')::numeric, 0) AS fisica_cop,
       ROUND(AVG(f.ingreso_neto_mensual_cop) FILTER (
             WHERE p.tipo_plataforma = 'En linea / Basada en la Nube')::numeric, 0) AS en_linea_cop
FROM fact_gig_economy f
JOIN dim_tiempo ti ON ti.id_tiempo = f.id_tiempo
JOIN dim_plataforma p ON p.id_plataforma = f.id_plataforma
GROUP BY ti.anio
ORDER BY ti.anio;
