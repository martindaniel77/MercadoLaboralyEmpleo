VISTA 4 — ANÁLISIS MULTIVARIADO: PERFILES, RETORNO Y VULNERABILIDAD

Pregunta: ¿Qué combinaciones de nivel educativo, modelo de plataforma y horas
dedicadas explican los mayores ingresos netos, y cómo se relaciona esto con la
dependencia económica y la cotización a pensión?

Dimensiones: dim_trabajador (nivel_educativo, dependencia_ingresos,
             afiliacion_pension)
             dim_plataforma (tipo_plataforma)
Medidas:     AVG(ingreso_neto_mensual_cop), AVG(ingreso_neto_hora_usd),
             proporción de cotización a pensión.

1. INDICADORES GLOBALES (KPIs)

SELECT ROUND(AVG(f.ingreso_neto_mensual_cop)::numeric, 0)             AS ingreso_neto_promedio_cop,
       COUNT(DISTINCT f.id_trabajador) FILTER (
           WHERE t.afiliacion_pension = 'Cotiza activamente')         AS cotizan,
       COUNT(DISTINCT f.id_trabajador)                                AS total,
       ROUND(100.0 * COUNT(DISTINCT f.id_trabajador) FILTER (
             WHERE t.afiliacion_pension = 'Cotiza activamente') /
             COUNT(DISTINCT f.id_trabajador), 1)                      AS pct_cotizan
FROM fact_gig_economy f
JOIN dim_trabajador t ON t.id_trabajador = f.id_trabajador;

2. INGRESO NETO PROMEDIO POR NIVEL EDUCATIVO x MODELO DE PLATAFORMA
SELECT t.nivel_educativo,
       p.tipo_plataforma,
       ROUND(AVG(f.ingreso_neto_mensual_cop)::numeric, 0) AS ingreso_promedio_cop,
       ROUND(AVG(f.ingreso_neto_hora_usd)::numeric, 2)    AS tarifa_promedio_usd_h,
       COUNT(DISTINCT f.id_trabajador)                    AS trabajadores
FROM fact_gig_economy f
JOIN dim_trabajador t ON t.id_trabajador = f.id_trabajador
JOIN dim_plataforma p ON p.id_plataforma = f.id_plataforma
GROUP BY t.nivel_educativo, p.tipo_plataforma
ORDER BY t.nivel_educativo, p.tipo_plataforma;

3. INGRESO NETO PROMEDIO POR NIVEL EDUCATIVO (agregado, para ver el gradiente)
SELECT t.nivel_educativo,
       ROUND(AVG(f.ingreso_neto_mensual_cop)::numeric, 0) AS ingreso_promedio_cop
FROM fact_gig_economy f
JOIN dim_trabajador t ON t.id_trabajador = f.id_trabajador
GROUP BY t.nivel_educativo
ORDER BY ingreso_promedio_cop DESC;

4. ROLL-UP: INGRESO NETO PROMEDIO POR GRUPO FORMATIVO
SELECT CASE
           WHEN t.nivel_educativo IN ('Primaria', 'Secundaria / Bachillerato') THEN 'Basica'
           WHEN t.nivel_educativo = 'Tecnico / Tecnologo'                     THEN 'Tecnica'
           ELSE 'Superior'
       END                                             AS grupo_formativo,
       ROUND(AVG(f.ingreso_neto_mensual_cop)::numeric, 0) AS ingreso_promedio_cop,
       COUNT(*)                                        AS trabajadores
FROM fact_gig_economy f
JOIN dim_trabajador t ON t.id_trabajador = f.id_trabajador
GROUP BY grupo_formativo
ORDER BY ingreso_promedio_cop;


5. TARIFA USD/H POR BANDA DE JORNADA SEMANAL (rendimientos decrecientes)
SELECT CASE
           WHEN f.horas_semanales < 30 THEN '< 30 h'
           WHEN f.horas_semanales < 40 THEN '30-40 h'
           WHEN f.horas_semanales < 48 THEN '40-48 h'
           WHEN f.horas_semanales < 56 THEN '48-56 h'
           ELSE '> 56 h'
       END                                               AS banda_jornada,
       p.tipo_plataforma,
       ROUND(AVG(f.ingreso_neto_hora_usd)::numeric, 2)   AS tarifa_promedio_usd_h,
       ROUND(AVG(f.horas_semanales)::numeric, 1)         AS horas_promedio,
       COUNT(DISTINCT f.id_trabajador)                   AS trabajadores
FROM fact_gig_economy f
JOIN dim_plataforma p ON p.id_plataforma = f.id_plataforma
GROUP BY banda_jornada, p.tipo_plataforma
ORDER BY MIN(f.horas_semanales), p.tipo_plataforma;

6. COTIZACIÓN A PENSIÓN SEGÚN DEPENDENCIA ECONÓMICA
SELECT t.dependencia_ingresos,
       COUNT(DISTINCT f.id_trabajador) AS total,
       COUNT(DISTINCT f.id_trabajador) FILTER (
           WHERE t.afiliacion_pension = 'Cotiza activamente') AS cotizan,
       ROUND(100.0 * COUNT(DISTINCT f.id_trabajador) FILTER (
             WHERE t.afiliacion_pension = 'Cotiza activamente') /
             COUNT(DISTINCT f.id_trabajador), 1)               AS pct_cotizan
FROM fact_gig_economy f
JOIN dim_trabajador t ON t.id_trabajador = f.id_trabajador
GROUP BY t.dependencia_ingresos
ORDER BY pct_cotizan DESC;

7. DESPROTECCIÓN DE LOS DEPENDIENTES EXCLUSIVOS POR MODELO DE PLATAFORMA
SELECT p.tipo_plataforma,
       COUNT(DISTINCT f.id_trabajador) FILTER (
           WHERE t.dependencia_ingresos = 'Unica fuente de ingresos (100%)') AS dependientes_unicos,
       COUNT(DISTINCT f.id_trabajador) FILTER (
           WHERE t.dependencia_ingresos = 'Unica fuente de ingresos (100%)'
             AND t.afiliacion_pension = 'No cotiza')                          AS unicos_no_cotizan,
       ROUND(100.0 * COUNT(DISTINCT f.id_trabajador) FILTER (
             WHERE t.dependencia_ingresos = 'Unica fuente de ingresos (100%)'
               AND t.afiliacion_pension = 'No cotiza') /
             COUNT(DISTINCT f.id_trabajador) FILTER (
             WHERE t.dependencia_ingresos = 'Unica fuente de ingresos (100%)'), 1) AS pct_no_cotizan
FROM fact_gig_economy f
JOIN dim_trabajador t ON t.id_trabajador = f.id_trabajador
JOIN dim_plataforma p ON p.id_plataforma = f.id_plataforma
GROUP BY p.tipo_plataforma;

8. MATRIZ: NIVEL EDUCATIVO x MODELO DE PLATAFORMA (medida = ingreso neto COP)
SELECT t.nivel_educativo,
       ROUND(AVG(f.ingreso_neto_mensual_cop) FILTER (
             WHERE p.tipo_plataforma = 'Fisica / Basada en Ubicacion')::numeric, 0) AS fisica_cop,
       ROUND(AVG(f.ingreso_neto_mensual_cop) FILTER (
             WHERE p.tipo_plataforma = 'En linea / Basada en la Nube')::numeric, 0) AS en_linea_cop
FROM fact_gig_economy f
JOIN dim_trabajador t ON t.id_trabajador = f.id_trabajador
JOIN dim_plataforma p ON p.id_plataforma = f.id_plataforma
GROUP BY t.nivel_educativo
ORDER BY t.nivel_educativo;
